static void IRAM_ATTR timer_callback(void *arg)
{
}

void init_timer()
{
  const esp_timer_create_args_t timer_args = {
      .callback = &timer_callback,
      .name = "speed_timer"};
  esp_timer_handle_t speed_timer;
  ESP_ERROR_CHECK(esp_timer_create(&timer_args, &speed_timer));
  ESP_ERROR_CHECK(esp_timer_start_periodic(speed_timer, TIMER_INTERVAL_US));
}

static TimerHandle_t xTimer1 = NULL;

static void end_of_manouver_callback(TimerHandle_t xTimer)
{
  p_controler_enabled_flag = true;
  junction_detection_enabled_flag = true;
  pwm_real_A = PWM_BASE_VALUE;
  pwm_real_B = PWM_BASE_VALUE;
}

void init_my_timer(void)
{
  xTimer1 = xTimerCreate(
      "Timer1",
      pdMS_TO_TICKS(1000),
      pdFALSE,
      (void *)0,
      end_of_manouver_callback);
}

void turn(int degrees)
{
  goal_steps_A = abs((10 * 314 * degrees * 130) / (100 * 100 * 360));
  goal_steps_B = abs((10 * 314 * degrees * 130) / (100 * 100 * 360));

  p_manoeuvre_controler_enabled_flag = true;
  step_check_enabled_flag = true;

  clear_both_encoders_count();

  base_speed_pwm_A = degrees > 0 ? 500 : -500;
  base_speed_pwm_B = degrees > 0 ? -500 : 500;

  p_line_controler_enabled_flag = false;
  junction_detection_enabled_flag = false;
}
